function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_RAW") || ss.insertSheet("DB_RAW");
  dbSheet.hideSheet();
  var raw = dbSheet.getRange("A1").getValue() || "{}";

  // Jika dipanggil oleh fail HTML luar menggunakan JSONP
  if (e && e.parameter && e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + "(" + raw + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  // Jika dibuka terus sebagai Web App di telefon / komputer (Auto-Sync 100%)
  try {
    var html = HtmlService.createHtmlOutputFromFile("Index")
      .setTitle("ArifOps Pro™ | Operasi & Pengurusan Aset")
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag("viewport", "width=device-width, initial-scale=1.0");
    return html;
  } catch (err) {
    return ContentService.createTextOutput(raw).setMimeType(ContentService.MimeType.JSON);
  }
}

function getAppData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_RAW") || ss.insertSheet("DB_RAW");
  dbSheet.hideSheet();
  return dbSheet.getRange("A1").getValue() || "{}";
}

function saveAppData(jsonStr) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_RAW") || ss.insertSheet("DB_RAW");
  dbSheet.hideSheet();
  dbSheet.getRange("A1").setValue(jsonStr);

  try {
    var data = JSON.parse(jsonStr);
    var viewSheet = ss.getSheets()[0];
    viewSheet.clear();
    var v = data.vehicle || {};
    var brand = data.brand || {};
    viewSheet.appendRow([brand.title || "ArifOps Pro", brand.owner || "Arif • Operasi & Pengurusan Aset"]);
    viewSheet.appendRow(["Kenderaan", v.model || "", "Unit", v.plateNo || "", "Status", v.overallStatus || ""]);
    viewSheet.appendRow(["Roadtax", v.roadtaxExpiry || "", "Insurans", v.insuranceExpiry || "", "Puspakom", v.puspakomExpiry || ""]);
    viewSheet.appendRow(["Lesen Pemandu", v.driverLicenseExpiry || "", "Odometer Semasa", v.currentOdometerKm || "", "Target Servis", v.nextServiceKm || ""]);
    viewSheet.appendRow([]);
    viewSheet.appendRow(["Kategori", "Nama Peralatan / Ubat", "Disemak", "Kuantiti", "Status", "Tarikh Luput", "Catatan"]);
    if (Array.isArray(data.equipment)) {
      data.equipment.forEach(function(item) {
        viewSheet.appendRow([item.category, item.name, item.checked ? "Ya" : "Tidak", item.qty, item.condition, item.expiry || "-", item.note || "-"]);
      });
    }
  } catch (err) {}
  return "OK";
}

function doPost(e) {
  var jsonStr = (e.parameter && e.parameter.payload) ? e.parameter.payload : (e.postData ? e.postData.contents : "{}");
  saveAppData(jsonStr);
  return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT);
}
